//GitHub Actions workflow file
mkdir -p .github/workflows
nano .github/workflows/deploy.yml

#yaml file me likho
name: Deploy to EC2 Meditation App

on:
  push:
    branches:
      - main
  workflow_dispatch:

jobs:
  deploy:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Deploy to EC2
        uses: appleboy/ssh-action@v1
        with:
          host: 52.66.106.247
          username: ubuntu
          key: ${{ secrets.EC2_SSH_KEY }}
          port: 22
          script: |
            cd /home/ubuntu/meditation
            git pull origin main
            
            # Sirf tab install karo jab package-lock.json change ho
            if [ "package-lock.json" -nt "node_modules" ]; then
              echo "Installing dependencies..."
              npm ci --production
            else
              echo "✅ Dependencies up to date, skipping install"
            fi
            
            pm2 restart Meditation


//Commit and push karo
git add .github/workflows/deploy.yml
git commit -m "Add GitHub Actions workflow for EC2 deployment"
git push origin main
